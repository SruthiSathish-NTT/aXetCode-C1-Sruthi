  ~~~~~~~~~~~~
  Command-line entry point.  Subcommands are wired in later milestones.
  """

  from __future__ import annotations

  import argparse
  import sys


  def build_parser() -> argparse.ArgumentParser:
      parser = argparse.ArgumentParser(
          prog="logscope",
          description="Streaming analysis of application logs.",
      )
      sub = parser.add_subparsers(dest="command", metavar="COMMAND")

      # Placeholder registrations — implementations added in M2-M4.
      for name in ("summary", "timeline", "grep"):
          sp = sub.add_parser(name, help=f"{name} (coming soon)")
          sp.add_argument("file", metavar="LOG_FILE")
          sp.add_argument("--format", choices=["json", "md"], default=None)
          if name == "timeline":
              sp.add_argument("--bucket", default="15m")
          if name == "grep":
              sp.add_argument("--level", required=True)
              sp.add_argument("--since", default=None)

      return parser


  def main(argv: list[str] | None = None) -> None:
      parser = build_parser()
      args = parser.parse_args(argv)

          parser.print_help()
          sys.exit(0)

      print(f"[stub] command '{args.command}' not yet implemented.")
      sys.exit(1)


  if __name__ == "__main__":

####  logscope/parser.py

  """
  logscope.parser
  ~~~~~~~~~~~~~~~
  Streaming log parser.  Reads one line at a time; never loads the full file.

  Public API
  ----------
  LogEntry      – dataclass for a successfully parsed line
  MalformedLine – dataclass for an unparseable line
  parse_line    – parse a single raw string
  iter_log      – yield LogEntry | MalformedLine for every line in a file
  """

  from __future__ import annotations

  import re
  from dataclasses import dataclass
  from datetime import datetime
  from typing import Iterator

  # Valid log levels defined by the spec.
  VALID_LEVELS: frozenset[str] = frozenset({"INFO", "WARN", "ERROR"})

  _SPLIT_RE = re.compile(r"\S+")


  @dataclass(frozen=True, slots=True)
  class LogEntry:
      """A successfully parsed log line."""

      timestamp: datetime
      level: str
      message: str
      raw: str  # original untouched line (stripped of trailing newline)


  @dataclass(frozen=True, slots=True)
  class MalformedLine:
      """A line that could not be parsed."""

      raw: str


  def parse_line(line: str) -> LogEntry | MalformedLine:
      """Parse *line* and return a LogEntry or MalformedLine."""
      raw = line.rstrip("\n\r")

      parts = raw.split(None, 3)  # split on any whitespace, at most 3 splits
      if len(parts) < 4:
          return MalformedLine(raw=raw)

      date_str, time_str, level, message = parts

      if level not in VALID_LEVELS:
          return MalformedLine(raw=raw)

      try:
          timestamp = datetime.strptime(f"{date_str} {time_str}", "%Y-%m-%d %H:%M:%S")
      except ValueError:
          return MalformedLine(raw=raw)

      return LogEntry(timestamp=timestamp, level=level, message=message, raw=raw)


  def iter_log(path: str) -> Iterator[LogEntry | MalformedLine]:
      """Yield one LogEntry or MalformedLine per line in the file at *path*."""
      with open(path, "r", encoding="utf-8", errors="replace") as fh:
          for line in fh:  # one line at a time — no read() / readlines()
              yield parse_line(line)

### 5. Streaming and malformed-line guarantees

Streaming:  iter_log  opens the file and iterates with  for line in fh  — Python's standard file iterator, which yields
one line at a time from the OS buffer.  read()  and  readlines()  are never called. Memory use is O(1) relative to file
size; only the current line is held.

Malformed lines:  parse_line  returns a  MalformedLine  sentinel for every failure path (too few tokens, unknown level,
bad date/time). There is no bare  except  — only  ValueError  from  strptime  is caught. All other errors propagate
normally (e.g. file-not-found), but no malformed log content can crash the parser. This is verified by
test_malformed_line_does_not_crash  (100 garbage lines, zero exceptions) and  test_non_utf8_bytes_do_not_crash  (binary
garbage in the file).
