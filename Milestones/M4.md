####  logscope/commands/grep.py

  """
  logscope.commands.grep
  ~~~~~~~~~~~~~~~~~~~~~~
  Single-pass streaming grep over a log file.

  Public API
  ----------
  GrepResult      – dataclass holding matched lines and malformed count
  VALID_LEVELS    – frozenset of accepted level strings
  parse_since     – parse an optional --since string → datetime | None
  run             – consume iter_log and collect results
  """

  from __future__ import annotations

  from dataclasses import dataclass
  from datetime import datetime

  from logscope.parser import LogEntry, MalformedLine, iter_log

  VALID_LEVELS: frozenset[str] = frozenset({"INFO", "WARN", "ERROR"})

  _SINCE_FMT = "%Y-%m-%d %H:%M"


  def parse_since(value: str) -> datetime:
      """Parse a --since string (YYYY-MM-DD HH:MM) and return a datetime.

      Raises ValueError if the format does not match exactly.
      """
      try:
          return datetime.strptime(value, _SINCE_FMT)
      except ValueError:
          raise ValueError(
              f"Invalid --since value {value!r}. "
              "Expected format: YYYY-MM-DD HH:MM  (e.g. \"2026-10-01 09:00\")."
          )


  @dataclass
  class GrepResult:
      """Result produced by the grep command."""

      lines: list[str]   # original raw log lines that matched (no trailing newline)
      malformed: int


  def run(path: str, level: str, since: datetime | None = None) -> GrepResult:
      """Stream *path* and return a GrepResult.

      Raises ValueError for an invalid level (before opening the file).
      Raises OSError / FileNotFoundError if the file cannot be opened.
      """
      if level not in VALID_LEVELS:
          raise ValueError(
              f"Invalid level {level!r}. Must be one of: "
              + ", ".join(sorted(VALID_LEVELS))
              + "."
          )

      matched: list[str] = []
      malformed: int = 0

      for entry in iter_log(path):
          if isinstance(entry, MalformedLine):
              malformed += 1
              continue
          if entry.level != level:
              continue
          if since is not None and entry.timestamp < since:
              continue
          matched.append(entry.raw)

      return GrepResult(lines=matched, malformed=malformed)

  def format_grep(result: GrepResult, level: str, since_str: str | None) -> str:
      """Return a human-readable grep report."""
      out: list[str] = []
      since_note = f"  since : {since_str}" if since_str else ""
      out.append(f"=== Grep results (level: {level}){since_note} ===")
      out.append("")
      if result.lines:
          for raw_line in result.lines:
              out.append(raw_line)
      else:
          out.append("  (no matching lines)")
      out.append("")
      out.append(f"Matches         : {len(result.lines)}")
      out.append(f"Malformed lines : {result.malformed}")
      return "\n".join(out)

### 5. Where grep logic lives and why it stays streaming

All filtering logic lives in  logscope/commands/grep.py  in  run() . It calls  iter_log(path)  — the same generator from
M1 — and evaluates one entry at a time in a  for  loop. There is no intermediate buffer; each entry is tested against
level  and  since  and either appended to  matched  or discarded immediately. Memory is O(matching lines), not O(file
size). For a  --level ERROR  run on a mostly-INFO log the matched list is small; in the worst case (all lines match) it
equals the size of the matching lines' text, which the spec accepts because  grep  must return matching lines.

The  --since  comparison  entry.timestamp < since  is a single  datetime  subtraction on an already-parsed value — no
re-parsing, no buffering.
