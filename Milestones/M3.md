  """
  logscope.commands.timeline
  ~~~~~~~~~~~~~~~~~~~~~~~~~~
  Single-pass streaming ERROR timeline bucketed by configurable time interval.

  Public API
  ----------
  TimelineResult  – dataclass holding all aggregated data
  parse_bucket    – parse a bucket string such as "15m" or "1h" → timedelta
  bucket_key      – floor a datetime to the nearest bucket boundary
  run             – consume iter_log and return a TimelineResult
  """

  from __future__ import annotations

  import re
  from collections import defaultdict
  from dataclasses import dataclass
  from datetime import datetime, timedelta

  from logscope.parser import LogEntry, MalformedLine, iter_log

  _EPOCH = datetime(1970, 1, 1)
  _BUCKET_RE = re.compile(r"^(\d+)(m|h)$")


  def parse_bucket(spec: str) -> timedelta:
      """Parse a bucket specification string and return a timedelta.

      Accepted formats: ``<N>m`` (minutes), ``<N>h`` (hours).
      Raises ValueError for anything else or for N == 0.
      """
      m = _BUCKET_RE.match(spec.strip())
      if not m:
          raise ValueError(
              f"Invalid bucket specification {spec!r}. "
              "Expected a positive integer followed by 'm' (minutes) or 'h' (hours), "
              "e.g. '15m' or '1h'."
          )
      value = int(m.group(1))
      unit = m.group(2)
      if value == 0:
          raise ValueError(
              f"Bucket size must be greater than zero, got {spec!r}."
          )
      if unit == "m":
          return timedelta(minutes=value)
      return timedelta(hours=value)


  def bucket_key(ts: datetime, bucket_td: timedelta) -> datetime:
      """Return the bucket-start datetime for *ts* given bucket width *bucket_td*.

      Floors elapsed seconds since the UNIX epoch to the nearest multiple of
      *bucket_td*, producing stable boundaries (e.g. every 15 min on the hour).
      """
      delta = ts - _EPOCH
      bucket_seconds = int(bucket_td.total_seconds())
      elapsed_seconds = int(delta.total_seconds())
      floored_seconds = (elapsed_seconds // bucket_seconds) * bucket_seconds
      return _EPOCH + timedelta(seconds=floored_seconds)


  @dataclass
  class TimelineResult:
      """All data produced by the timeline command."""

      buckets: dict[datetime, int]   # bucket start datetime → error count
      malformed: int
      bucket_label: str              # e.g. "15m"


  def run(path: str, bucket_spec: str) -> TimelineResult:
      """Stream *path* once and return a TimelineResult.

      Only ERROR-level entries are counted.  INFO and WARN are discarded in the
      loop body; the file is still consumed one line at a time via iter_log, so
      memory use is O(distinct bucket count), not O(file size).

      Raises ValueError for an invalid bucket_spec (before opening the file).
      Raises OSError / FileNotFoundError if the file cannot be opened.
      """
      bucket_td = parse_bucket(bucket_spec)   # raises ValueError early if bad

      raw_buckets: defaultdict[datetime, int] = defaultdict(int)
      malformed: int = 0

      for entry in iter_log(path):
          if isinstance(entry, MalformedLine):
              malformed += 1
              continue
          if entry.level != "ERROR":
              continue
          key = bucket_key(entry.timestamp, bucket_td)
          raw_buckets[key] += 1

      sorted_buckets: dict[datetime, int] = dict(sorted(raw_buckets.items()))

      return TimelineResult(
          buckets=sorted_buckets,
          malformed=malformed,
          bucket_label=bucket_spec.strip(),
      )

####  logscope/output/text.py

  """
  logscope.output.text
  ~~~~~~~~~~~~~~~~~~~~
  Human-readable terminal renderers.  Each function accepts the relevant
  result dataclass and returns a formatted string.  Nothing is printed here;
  callers decide how to emit the output (print, write to file, etc.).
  """

  from __future__ import annotations

  import shutil

  from logscope.commands.summary import SummaryResult
  from logscope.commands.timeline import TimelineResult

  _TS_FMT = "%Y-%m-%d %H:%M:%S"
  _DEFAULT_BAR_WIDTH = 40
  _RESERVED_COLS = 29


  def format_summary(result: SummaryResult) -> str:
      """Return a multi-line human-readable summary report."""
      lines: list[str] = []
      lines.append("=== Log Summary ===")
      lines.append("")
      lines.append("Line counts:")
      lines.append(f"  INFO   : {result.counts['INFO']}")
      lines.append(f"  WARN   : {result.counts['WARN']}")
      lines.append(f"  ERROR  : {result.counts['ERROR']}")
      lines.append("")
      first = result.first_ts.strftime(_TS_FMT) if result.first_ts else "—"
      last  = result.last_ts.strftime(_TS_FMT)  if result.last_ts  else "—"
      lines.append(f"First timestamp : {first}")
      lines.append(f"Last timestamp  : {last}")
      lines.append("")
      lines.append("Top ERROR messages:")
      if result.top_errors:
          for msg, count in result.top_errors:
              lines.append(f"  [{count:>5}]  {msg}")
      else:
          lines.append("  (none)")
      lines.append("")
      lines.append(f"Malformed lines : {result.malformed}")
      return "\n".join(lines)


  def format_timeline(result: TimelineResult) -> str:
      """Return a multi-line bar-chart timeline report."""
      lines: list[str] = []
      lines.append(f"=== Error Timeline (bucket: {result.bucket_label}) ===")
      lines.append("")

      if not result.buckets:
          lines.append("  (no ERROR records found)")
      else:
          try:
              term_cols = shutil.get_terminal_size(fallback=(0, 0)).columns
          except Exception:
              term_cols = 0
          available = term_cols - _RESERVED_COLS if term_cols > _RESERVED_COLS else 0
          max_bar = max(available, _DEFAULT_BAR_WIDTH)
          bar_width = min(max_bar, _DEFAULT_BAR_WIDTH)
          max_count = max(result.buckets.values())

          for bucket_start, count in result.buckets.items():
              ts_str = bucket_start.strftime(_TS_FMT)
              bar_len = round((count / max_count) * bar_width) if max_count > 0 else 0
              bar = "█" * bar_len
              lines.append(f"{ts_str}  [{count:>5}]  {bar}")

      lines.append("")
      lines.append(f"Malformed lines : {result.malformed}")
      return "\n".join(lines)

####  logscope/cli.py

  """
  logscope.cli
  ~~~~~~~~~~~~
  Command-line entry point.
  """

  from __future__ import annotations

  import argparse
  import sys


  def build_parser() -> argparse.ArgumentParser:
      parser = argparse.ArgumentParser(
          prog="logscope",
          description="Streaming analysis of application logs.",
      )
      sub = parser.add_subparsers(dest="command", metavar="COMMAND")

      sp_summary = sub.add_parser("summary", help="Summarise a log file.")
      sp_summary.add_argument("file", metavar="LOG_FILE")
      sp_summary.add_argument("--format", choices=["json", "md"], default=None)

      sp_timeline = sub.add_parser("timeline", help="Show ERROR timeline bucketed by interval.")
      sp_timeline.add_argument("file", metavar="LOG_FILE")
      sp_timeline.add_argument("--bucket", default="15m", metavar="BUCKET",
                               help="Bucket duration, e.g. 15m or 1h (default: 15m).")
      sp_timeline.add_argument("--format", choices=["json", "md"], default=None)

      sp_grep = sub.add_parser("grep", help="grep (coming soon)")
      sp_grep.add_argument("file", metavar="LOG_FILE")
      sp_grep.add_argument("--level", required=True)
      sp_grep.add_argument("--since", default=None)
      sp_grep.add_argument("--format", choices=["json", "md"], default=None)

      return parser


  def main(argv: list[str] | None = None) -> None:
      parser = build_parser()
      args = parser.parse_args(argv)
      if args.command is None:
          parser.print_help()
          sys.exit(0)
      if args.command == "summary":
          _run_summary(args)
      elif args.command == "timeline":
          _run_timeline(args)
      else:
          print(f"[stub] command '{args.command}' not yet implemented.")
          sys.exit(1)


  def _run_summary(args: argparse.Namespace) -> None:
      from logscope.commands.summary import run
      from logscope.output.text import format_summary
      try:
          result = run(args.file)
      except FileNotFoundError:
          print(f"logscope: error: file not found: {args.file}", file=sys.stderr)
          sys.exit(2)
      except OSError as exc:
          print(f"logscope: error: cannot read file: {exc}", file=sys.stderr)
          sys.exit(2)
      if args.format is not None:
          print(f"logscope: --format {args.format} will be available in a later milestone.",
                file=sys.stderr)
          sys.exit(1)
      print(format_summary(result))


  def _run_timeline(args: argparse.Namespace) -> None:
      from logscope.commands.timeline import run, parse_bucket
      from logscope.output.text import format_timeline
      try:
          parse_bucket(args.bucket)
      except ValueError as exc:
          print(f"logscope: error: {exc}", file=sys.stderr)
          sys.exit(2)
      try:
          result = run(args.file, args.bucket)
      except FileNotFoundError:
          print(f"logscope: error: file not found: {args.file}", file=sys.stderr)
          sys.exit(2)
      except OSError as exc:
          print(f"logscope: error: cannot read file: {exc}", file=sys.stderr)
          sys.exit(2)
      if args.format is not None:
          print(f"logscope: --format {args.format} will be available in a later milestone.",
                file=sys.stderr)
          sys.exit(1)
      print(format_timeline(result))


  if __name__ == "__main__":
      main()

### 5. Where the timeline aggregation logic lives and why it stays streaming

All aggregation lives in  logscope/commands/timeline.py  in the  run()  function. The function calls  iter_log(path)  —
the same generator from M1 — and processes one  LogEntry | MalformedLine  at a time in a  for  loop. At no point is the
log read into a list or buffer. The only data structure that grows with input is  raw_buckets  (a  defaultdict ), which
holds at most one entry per distinct bucket boundary, not one entry per log line. For a 100 MB log with 15-minute
buckets over a 24-hour window that is at most 96 integers — O(1) in practice relative to file size.
